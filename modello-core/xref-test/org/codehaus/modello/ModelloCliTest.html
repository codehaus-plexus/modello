79      @Test
80      void testTypedParametersViaDefine() throws Exception {
81          String[] args = {
82              "--model",
83              "m.mdo",
84              "--type",
85              "java",
86              "--dir",
87              "out",
88              "--version",
89              "1.0.0",
90              "-DvelocityParameters=a=1,b=2",
91              "-Dmodello.plural.exceptions=aliases=alias",
92              "-DlicenseText=L1\nL2"
93          };
94          ModelloCli.parseArguments(args);
95          Map<String, Object> params = ModelloCli.getParameters();
96          Map<String, String> velocity = new HashMap<>();
97          velocity.put("a", "1");
98          velocity.put("b", "2");
99          assertEquals(velocity, params.get(ModelloParameterConstants.VELOCITY_PARAMETERS));
100         assertEquals(
101                 Collections.singletonMap("aliases", "alias"), params.get(ModelloParameterConstants.PLURAL_EXCEPTIONS));
102         assertEquals(Arrays.asList("L1", "L2"), params.get(ModelloParameterConstants.LICENSE_TEXT));
103     }

169     @Test
170     void testNamedCamelCaseOptions() throws Exception {
171         String[] args = {
172             "--modelFile",
173             "model.mdo",
174             "--outputType",
175             "xpp3-reader",
176             "--outputDirectory",
177             "target/out",
178             "--modelVersion",
179             "1.1.0",
180             "--packageWithVersion",
181             "--javaSource",
182             "11",
183             "--domAsXpp3",
184             "true",
185             "--velocityBasedir",
186             "templates",
187             "--templates",
188             "t1.vm,t2.vm",
189             "--params",
190             "k1=v1,k2=v2",
191             "--xsdFileName",
192             "model.xsd",
193             "--jsonSchemaFileName",
194             "model.json",
195             "--enforceMandatoryElements",
196             "--extendedClassnameSuffix",
197             "Ex",
198             "--firstVersion",
199             "1.0.0",
200             "--xdocFileName",
201             "model.xml",
202             "--pluralExceptions",
203             "kisses=kiss,parties=party"
204         };
205 
206         ModelloCli.parseArguments(args);
207 
208         assertEquals(new File("model.mdo"), ModelloCli.getModelFile());
209         assertEquals("xpp3-reader", ModelloCli.getOutputType());
210         Map<String, Object> params = ModelloCli.getParameters();
211         assertEquals("target/out", params.get(ModelloParameterConstants.OUTPUT_DIRECTORY));
212         assertEquals("1.1.0", params.get(ModelloParameterConstants.VERSION));
213         assertEquals("true", params.get(ModelloParameterConstants.PACKAGE_WITH_VERSION));
214         assertEquals("11", params.get(ModelloParameterConstants.OUTPUT_JAVA_SOURCE));
215         assertEquals("true", params.get(ModelloParameterConstants.DOM_AS_XPP3));
216         assertEquals("templates", params.get(ModelloParameterConstants.VELOCITY_BASEDIR));
217         assertEquals("t1.vm,t2.vm", params.get(ModelloParameterConstants.VELOCITY_TEMPLATES));
218         assertEquals("model.xsd", params.get(ModelloParameterConstants.OUTPUT_XSD_FILE_NAME));
219         assertEquals("model.json", params.get(ModelloParameterConstants.OUTPUT_JSONSCHEMA_FILE_NAME));
220         assertEquals("true", params.get(ModelloParameterConstants.XSD_ENFORCE_MANDATORY_ELEMENTS));
221         assertEquals("Ex", params.get(ModelloParameterConstants.EXTENDED_CLASSNAME_SUFFIX));
222         assertEquals("1.0.0", params.get(ModelloParameterConstants.FIRST_VERSION));
223         assertEquals("model.xml", params.get(ModelloParameterConstants.OUTPUT_XDOC_FILE_NAME));
224 
225         @SuppressWarnings("unchecked")
226         Map<String, String> vParams = (Map<String, String>) params.get(ModelloParameterConstants.VELOCITY_PARAMETERS);
227         assertEquals("v1", vParams.get("k1"));
228 
229         @SuppressWarnings("unchecked")
230         Map<String, String> plurals = (Map<String, String>) params.get(ModelloParameterConstants.PLURAL_EXCEPTIONS);
231         assertEquals("kiss", plurals.get("kisses"));
232         assertEquals("party", plurals.get("parties"));
233     }

352     @Test
353     void testLicenseFileAndText(@TempDir Path tempDir) throws Exception {
354         Path licPath = tempDir.resolve("LICENSE.txt");
355         Files.write(licPath, Arrays.asList("Line 1", "Line 2"));
356 
357         String[] args = {
358             "--model", "model.mdo",
359             "--type", "java",
360             "--dir", "out",
361             "--version", "1.0.0",
362             "--licenseFile", licPath.toString()
363         };
364 
365         ModelloCli.parseArguments(args);
366 
367         @SuppressWarnings("unchecked")
368         List<String> license = (List<String>) ModelloCli.getParameters().get(ModelloParameterConstants.LICENSE_TEXT);
369         assertNotNull(license);
370         assertEquals(2, license.size());
371         assertEquals("Line 1", license.get(0));
372         assertEquals("Line 2", license.get(1));
