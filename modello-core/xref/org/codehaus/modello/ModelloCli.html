118     private static void alias(String paramKey, String... flags) {
119         for (String flag : flags) {
120             PARAM_ALIASES.put(flag, paramKey);
121         }
122     }
123 
124     private static void aliasBoolean(String paramKey, String... flags) {
125         alias(paramKey, flags);
126         BOOLEAN_OPTIONS.addAll(Arrays.asList(flags));
127     }

129     public static void main(String[] args) throws Exception {
130         Modello modello = new Modello();
131 
132         parseArgumentsFromCommandLine(args);
133 
134         for (String type : outputType.split(",")) {
135             type = type.trim();
136             if (!type.isEmpty()) {
137                 generate(modello, type);
138             }
139         }
140     }

142     // Mirrors AbstractModelloGeneratorMojo#doExecute: current version, then each packaged one in its own package
143     private static void generate(Modello modello, String type) throws Exception {
144         try (XmlStreamReader reader = new XmlStreamReader(modelFile)) {
145             modello.generate(reader, type, parameters);
146         }
147 
148         String allVersions = (String) parameters.get(ModelloParameterConstants.ALL_VERSIONS);
149         if (StringUtils.isEmpty(allVersions)) {
150             return;
151         }
152         Map<String, Object> versioned = new HashMap<>(parameters);
153         versioned.put(ModelloParameterConstants.PACKAGE_WITH_VERSION, Boolean.toString(true));
154         for (String version : allVersions.split(",")) {
155             version = version.trim();
156             if (!version.isEmpty()) {
157                 versioned.put(ModelloParameterConstants.VERSION, version);
158                 try (XmlStreamReader reader = new XmlStreamReader(modelFile)) {
159                     modello.generate(reader, type, versioned);
160                 }
161             }
162         }
163     }

165     public static void parseArgumentsFromCommandLine(String[] args) throws Exception {
166         try {
167             parseArguments(args);
168             checkModelFile();
169         } catch (IllegalArgumentException e) {
170             System.err.println(e.getMessage());
171             usage();
172             System.exit(1);
173         }
174     }

177     static void checkModelFile() {
178         if (!modelFile.isFile()) {
179             throw new IllegalArgumentException("Model file not found: " + modelFile.getAbsolutePath());
180         }
181     }

183     static void parseArguments(String[] args) throws Exception {
184         if (args == null || args.length == 0) {
185             throw new IllegalArgumentException("No arguments specified");
186         }
187 
188         if (args[0].startsWith("-")) {
189             parseNamedArguments(args);
190         } else {
191             parseLegacyArguments(args);
192         }
193     }

195     private static void parseNamedArguments(String[] args) throws Exception {
196         parameters = new HashMap<>();
197         modelFile = null;
198         outputType = null;
199 
200         for (int i = 0; i < args.length; i++) {
201             String arg = args[i];
202 
203             if ("-h".equals(arg) || "--help".equals(arg)) {
204                 usage();
205                 System.exit(0);
206             }
207 
208             if (arg.startsWith("-D")) {
209                 parseProperty(arg.substring(2));
210                 continue;
211             }
212 
213             if (!arg.startsWith("-")) {
214                 if (modelFile == null) {
215                     modelFile = new File(arg);
216                     continue;
217                 } else {
218                     throw new IllegalArgumentException("Unexpected argument: " + arg);
219                 }
220             }
221 
222             String key = arg;
223             String val = null;
224             int eq = arg.indexOf('=');
225             if (eq > 0) {
226                 key = arg.substring(0, eq);
227                 val = arg.substring(eq + 1);
228             }
229 
230             if (val == null) {
231                 if (BOOLEAN_OPTIONS.contains(key)) {
232                     if (i + 1 < args.length
233                             && ("true".equalsIgnoreCase(args[i + 1]) || "false".equalsIgnoreCase(args[i + 1]))) {
234                         val = args[++i];
235                     } else {
236                         val = "true";
237                     }
238                 } else {
239                     if (i + 1 >= args.length || args[i + 1].startsWith("-")) {
240                         throw new IllegalArgumentException("Missing value for option: " + key);
241                     }
242                     val = args[++i];
243                 }
244             }
245 
246             if ("--model".equals(key)
247                     || "--models".equals(key)
248                     || "--modelFile".equals(key)
249                     || "--model-file".equals(key)) {
250                 modelFile = new File(val);
251             } else if ("--type".equals(key)
252                     || "--outputType".equals(key)
253                     || "--output-type".equals(key)
254                     || "--generatorId".equals(key)
255                     || "--generator-id".equals(key)) {
256                 outputType = val;
257             } else if ("--licenseFile".equals(key) || "--license-file".equals(key)) {
258                 File lf = new File(val);
259                 if (!lf.isFile()) {
260                     throw new IllegalArgumentException("License file not found: " + val);
261                 }
262                 parameters.put(ModelloParameterConstants.LICENSE_TEXT, Files.readAllLines(lf.toPath()));
263             } else if (PARAM_ALIASES.containsKey(key)) {
264                 putParameter(PARAM_ALIASES.get(key), val, key);
265             } else {
266                 throw new IllegalArgumentException("Unknown option: " + key);
267             }
268         }
269 
270         validateNamedArguments();
271     }

273     private static Map<String, String> parseMap(String option, String val) {
274         Map<String, String> map = new HashMap<>();
275         for (String s : val.split(",")) {
276             if (s.trim().isEmpty()) {
277                 continue;
278             }
279             String[] e = s.split("=", 2);
280             if (e.length != 2 || e[0].trim().isEmpty()) {
281                 throw new IllegalArgumentException("Malformed key=value entry '" + s.trim() + "' in option: " + option);
282             }
283             if (map.putIfAbsent(e[0].trim(), e[1].trim()) != null) {
284                 throw new IllegalArgumentException("Duplicate key '" + e[0].trim() + "' in option: " + option);
285             }
286         }
287         return map;
288     }

290     private static void parseProperty(String prop) {
291         int eq = prop.indexOf('=');
292         if (eq <= 0) {
293             throw new IllegalArgumentException("Missing value in option: -D" + prop + " (use -D" + prop + "=value)");
294         }
295         String k = prop.substring(0, eq).trim();
296         String v = prop.substring(eq + 1).trim();
297         String option = "-D" + k;
298         if (PARAM_ALIASES.containsKey("--" + k)) {
299             putParameter(PARAM_ALIASES.get("--" + k), v, option);
300         } else {
301             putParameter(k, v, option);
302             if (!k.startsWith("modello.")) {
303                 putParameter("modello." + k, v, option);
304             }
305         }
306     }

308     // Generators cast these to Map / List, not String
309     private static void putParameter(String key, String val, String option) {
310         if (ModelloParameterConstants.VELOCITY_PARAMETERS.equals(key)
311                 || ModelloParameterConstants.PLURAL_EXCEPTIONS.equals(key)) {
312             parameters.put(key, parseMap(option, val));
313         } else if (ModelloParameterConstants.LICENSE_TEXT.equals(key)) {
314             parameters.put(key, Arrays.asList(val.split("\\r?\\n")));
315         } else {
316             parameters.put(key, val);
317         }
318     }

320     private static void validateNamedArguments() {
321         if (modelFile == null) {
322             throw new IllegalArgumentException(
323                     "Missing required parameter: model file (--model, --modelFile, --model-file)");
324         }
325         if (StringUtils.isEmpty(outputType)) {
326             throw new IllegalArgumentException(
327                     "Missing required parameter: output type (--type, --outputType, --output-type, --generatorId, --generator-id)");
328         }
329         if (!parameters.containsKey(ModelloParameterConstants.OUTPUT_DIRECTORY)
330                 || StringUtils.isEmpty((String) parameters.get(ModelloParameterConstants.OUTPUT_DIRECTORY))) {
331             throw new IllegalArgumentException(
332                     "Missing required parameter: output directory (--dir, --outputDirectory, --output-directory)");
333         }
334         if (!parameters.containsKey(ModelloParameterConstants.VERSION)
335                 || StringUtils.isEmpty((String) parameters.get(ModelloParameterConstants.VERSION))) {
336             throw new IllegalArgumentException(
337                     "Missing required parameter: model version (--version, --modelVersion, --model-version)");
338         }
339         if (!parameters.containsKey(ModelloParameterConstants.PACKAGE_WITH_VERSION)) {
340             parameters.put(ModelloParameterConstants.PACKAGE_WITH_VERSION, "false");
341         }
342         if (!parameters.containsKey(ModelloParameterConstants.OUTPUT_JAVA_SOURCE)) {
343             parameters.put(
344                     ModelloParameterConstants.OUTPUT_JAVA_SOURCE, ModelloParameterConstants.OUTPUT_JAVA_SOURCE_DEFAULT);
345         }
346     }

348     private static void parseLegacyArguments(String[] args) {
349         if (args.length < 6) {
350             throw new IllegalArgumentException("Insufficient arguments for legacy positional invocation");
351         }
352 
353         modelFile = new File(args[0]);
354 
355         outputType = args[1];
356 
357         parameters = new HashMap<>();
358 
359         String outputDirectory = args[2];
360 
361         if (StringUtils.isEmpty(outputDirectory)) {
362             throw new IllegalArgumentException("Missing required parameter: output directory");
363         }
364 
365         parameters.put(ModelloParameterConstants.OUTPUT_DIRECTORY, outputDirectory);
366 
367         String modelVersion = args[3];
368 
369         if (StringUtils.isEmpty(modelVersion)) {
370             throw new IllegalArgumentException("Missing required parameter: model version");
371         }
372 
373         parameters.put(ModelloParameterConstants.VERSION, modelVersion);
374 
375         String packageWithVersion = args[4];
376 
377         if (StringUtils.isEmpty(packageWithVersion)) {
378             throw new IllegalArgumentException("Missing required parameter: package with version");
379         }
380 
381         parameters.put(ModelloParameterConstants.PACKAGE_WITH_VERSION, packageWithVersion);
382 
383         String javaSource = args[5];
384 
385         if (StringUtils.isEmpty(javaSource)) {
386             throw new IllegalArgumentException("Missing required parameter: Java Source");
387         }
388 
389         parameters.put(ModelloParameterConstants.OUTPUT_JAVA_SOURCE, javaSource);
390 
391         if (args.length > 6) {
392             parameters.put(ModelloParameterConstants.ENCODING, args[6]);
393         }
394     }
