41  /**
42   * XsdMetadataPlugin
43   *
44   * @author <a href="mailto:joakim@erdfelt.com">Joakim Erdfelt</a>
45   */
46  @Named("xdoc")
47  @Singleton
48  public class XsdMetadataPlugin extends AbstractMetadataPlugin {
49      public static final String NAMESPACE = "xsd.namespace";
50  
51      public static final String TARGET_NAMESPACE = "xsd.targetNamespace";
52  
53      public static final String COMPOSITOR = "xsd.compositor";
54  
55      public static final String ENUMERATIONS = "xsd.enumerations";
56  
57      public static final String SUGGESTIONS = "xsd.suggestions";
58  
59      public static final String PATTERN = "xsd.pattern";
60  
61      public AssociationMetadata getAssociationMetadata(ModelAssociation association, Map<String, String> data)
62              throws ModelloException {
63          return new XsdAssociationMetadata();
64      }
65  
66      public ClassMetadata getClassMetadata(ModelClass clazz, Map<String, String> data) throws ModelloException {
67          XsdClassMetadata metadata = new XsdClassMetadata();
68  
69          metadata.setCompositor(getString(data, COMPOSITOR));
70  
71          return metadata;
72      }
73  
74      public InterfaceMetadata getInterfaceMetadata(ModelInterface iface, Map<String, String> data)
75              throws ModelloException {
76          return new XsdInterfaceMetadata();
77      }
78  
79      public FieldMetadata getFieldMetadata(ModelField field, Map<String, String> data) throws ModelloException {
80          XsdFieldMetadata metadata = new XsdFieldMetadata();
81  
82          List<String> enumerations = parseCommaSeparated(getString(data, ENUMERATIONS));
83          List<String> suggestions = parseCommaSeparated(getString(data, SUGGESTIONS));
84          String pattern = getString(data, PATTERN);
85  
86          long constraintCount = (enumerations.isEmpty() ? 0 : 1)
87                  + (suggestions.isEmpty() ? 0 : 1)
88                  + (pattern != null && !pattern.isEmpty() ? 1 : 0);
89          if (constraintCount > 1) {
90              throw new ModelloException("Field '"
91                      + field.getName()
92                      + "': xsd.enumerations, xsd.suggestions and xsd.pattern are mutually exclusive."
93                      + " Only one may be specified per field.");
94          }
95  
96          metadata.setEnumerations(enumerations);
97          metadata.setSuggestions(suggestions);
98          metadata.setPattern(pattern);
99  
100         return metadata;
101     }
102 
103     public ModelMetadata getModelMetadata(Model model, Map<String, String> data) throws ModelloException {
104         XsdModelMetadata metadata = new XsdModelMetadata();
105 
106         metadata.setNamespace(getString(data, NAMESPACE));
107 
108         metadata.setTargetNamespace(getString(data, TARGET_NAMESPACE));
109 
110         return metadata;
111     }
112 
113     private static List<String> parseCommaSeparated(String value) {
114         if (value == null || value.trim().isEmpty()) {
115             return Collections.emptyList();
116         }
117         return Arrays.stream(value.split(","))
118                 .map(String::trim)
119                 .filter(s -> !s.isEmpty())
120                 .collect(Collectors.toList());
121     }
122 }
