53  /**
54   * Tests that {@code xsd.enumerations}, {@code xsd.suggestions} and {@code xsd.pattern} field
55   * metadata attributes are correctly translated to XSD facets by {@link XsdGenerator}.
56   *
57   * @see <a href="https://github.com/codehaus-plexus/modello/issues/633">issue 633</a>
58   */
59  @PlexusTest
60  public class EnumerationXsdGeneratorTest extends AbstractModelloGeneratorTest {
61  
62      @Inject
63      private ModelloCore modello;
64  
65      public EnumerationXsdGeneratorTest() {
66          super("xsd-enum-test");
67      }
68  
69      @Test
70      public void testEnumerationConstraints() throws Throwable {
71          Model model = modello.loadModel(getXmlResourceReader("/xsd-enum-test.mdo"));
72  
73          Map<String, Object> parameters = getModelloParameters("1.0.0");
74          modello.generate(model, "xsd", parameters);
75  
76          File xsdFile = new File(getOutputDirectory(), "xsd-enum-test-1.0.0.xsd");
77          assertTrue(xsdFile.exists(), "Generated XSD file should exist: " + xsdFile);
78  
79          // ---- Structural assertions on the generated XSD ----
80  
81          DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
82          dbf.setNamespaceAware(true);
83          Document doc = dbf.newDocumentBuilder().parse(xsdFile);
84          XPath xpath = XPathFactory.newInstance().newXPath();
85  
86          // 1. scope must have a xs:restriction with xs:enumeration children
87          NodeList scopeEnums = (NodeList) xpath.evaluate(
88                  "//*[local-name()='element' and @name='scope']"
89                          + "//*[local-name()='restriction' and @base='xs:string']"
90                          + "/*[local-name()='enumeration']",
91                  doc,
92                  XPathConstants.NODESET);
93          assertEquals(5, scopeEnums.getLength(), "scope should have 5 strict enumerations");
94  
95          // 2. packaging must have a xs:union containing xs:enumeration members
96          NodeList packagingEnums = (NodeList) xpath.evaluate(
97                  "//*[local-name()='element' and @name='packaging']"
98                          + "//*[local-name()='union']"
99                          + "//*[local-name()='enumeration']",
100                 doc,
101                 XPathConstants.NODESET);
102         assertEquals(5, packagingEnums.getLength(), "packaging should have 5 enumeration suggestions inside a union");
103 
104         // 3. artifactId must have a xs:pattern facet
105         NodeList patternNodes = (NodeList) xpath.evaluate(
106                 "//*[local-name()='element' and @name='artifactId']" + "//*[local-name()='pattern']",
107                 doc,
108                 XPathConstants.NODESET);
109         assertEquals(1, patternNodes.getLength(), "artifactId should have exactly one xs:pattern facet");
110         assertEquals(
111                 "[a-zA-Z0-9._\\-]+",
112                 patternNodes.item(0).getAttributes().getNamedItem("value").getTextContent(),
113                 "artifactId pattern value should match");
114 
115         // 4. description must still be plain xs:string (no inline simpleType)
116         NodeList descType = (NodeList) xpath.evaluate(
117                 "//*[local-name()='element' and @name='description' and @type='xs:string']",
118                 doc,
119                 XPathConstants.NODESET);
120         assertEquals(1, descType.getLength(), "description should remain a plain xs:string element");
121 
122         // 5. attributeScope (xml.attribute="true") must have xs:restriction with 2 xs:enumeration children
123         //    — exercises the attribute-field loop in XsdGenerator
124         NodeList attributeScopeEnums = (NodeList) xpath.evaluate(
125                 "//*[local-name()='attribute' and @name='attributeScope']"
126                         + "//*[local-name()='restriction' and @base='xs:string']"
127                         + "/*[local-name()='enumeration']",
128                 doc,
129                 XPathConstants.NODESET);
130         assertEquals(
131                 2,
132                 attributeScopeEnums.getLength(),
133                 "attributeScope attribute should have 2 strict enumerations (compile, runtime)");
134 
135         // ---- Validation assertions ----
136 
137         SchemaFactory sf = SchemaFactory.newInstance(XMLConstants.W3C_XML_SCHEMA_NS_URI);
138         Schema schema = sf.newSchema(new StreamSource(xsdFile));
139         Validator validator = schema.newValidator();
140 
141         // Valid document passes
142         validator.validate(new StreamSource(getClass().getResourceAsStream("/xsd-enum-test-valid.xml")));
143 
144         // Invalid scope (strict enumeration) must be rejected
145         try {
146             validator.validate(new StreamSource(getClass().getResourceAsStream("/xsd-enum-test-invalid-scope.xml")));
147             fail("Validation should have failed for invalid scope value");
148         } catch (SAXParseException e) {
149             // expected
150             assertTrue(
151                     e.getMessage().contains("scope") || e.getMessage().contains("unknown-scope"),
152                     "Error should mention the invalid scope value: " + e.getMessage());
153         }
154 
155         // Invalid artifactId (pattern violation) must be rejected
156         try {
157             validator.validate(new StreamSource(getClass().getResourceAsStream("/xsd-enum-test-invalid-pattern.xml")));
158             fail("Validation should have failed for invalid artifactId pattern");
159         } catch (SAXParseException e) {
160             // expected
161             assertTrue(
162                     e.getMessage().contains("artifactId") || e.getMessage().contains("pattern"),
163                     "Error should mention the invalid artifactId value: " + e.getMessage());
164         }
165 
166         // Suggested value outside the enumeration (custom packaging type) must PASS validation
167         String customPackagingXml = "<?xml version=\"1.0\"?>"
168                 + "<artifact xmlns=\"http://codehaus-plexus.github.io/XSD-ENUM-TEST/1.0.0\">"
169                 + "<scope>test</scope>"
170                 + "<packaging>custom-type</packaging>"
171                 + "<artifactId>my-lib</artifactId>"
172                 + "</artifact>";
173         validator.validate(new StreamSource(new StringReader(customPackagingXml)));
174     }
